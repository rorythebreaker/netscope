  {
      Options options = Options.Parse(args);
      if (options.ShowHelp)
      {
          Console.WriteLine("usage: wc [--top N] [--ignore-case] <file>...");
          return 0;
      }
      if (options.Error != null)
      {
          Console.Error.WriteLine(options.Error);
          return 2;
      }
      WordCounter wordCounter = new WordCounter(options.IgnoreCase);
      foreach (string file in options.Files)
      {
          if (!File.Exists(file))
          {
              Console.Error.WriteLine("not found: " + file);
              return 3;
          }
          wordCounter.Add(File.ReadAllText(file));
      }
      foreach (var (value, value2) in wordCounter.Top(options.Top))
      {
          Console.WriteLine($"{value2,6} {value}");
      }
      return 0;
  }

  public static Options Parse(string[] args)
  {
      Options options = new Options();
      for (int i = 0; i < args.Length; i++)
      {
          switch (args[i])
          {
          case "-h":
          case "--help":
              options.ShowHelp = true;
              break;
          case "-i":
          case "--ignore-case":
              options.IgnoreCase = true;
              break;
          case "--top":
          {
              if (i + 1 >= args.Length || !int.TryParse(args[++i], out var result) || result <= 0)
              {
                  options.Error = "--top expects a positive number";
                  return options;
              }
              options.Top = result;
              break;
          }
          default:
              if (args[i].StartsWith("-", StringComparison.Ordinal))
              {
                  options.Error = "unknown option " + args[i];
                  return options;
              }
              options.Files.Add(args[i]);
              break;
          }
      }
      if (options.Files.Count == 0 && !options.ShowHelp)
      {
          options.Error = "no input files";
      }
      return options;
  }

  public void Add(string text)
  {
      int num = -1;
      for (int i = 0; i <= text.Length; i++)
      {
          bool flag = i < text.Length && char.IsLetter(text[i]);
          if (flag && num < 0)
          {
              num = i;
          }
          else if (!flag && num >= 0)
          {
              string key = text.Substring(num, i - num);
              _counts[key] = ((!_counts.TryGetValue(key, out var value)) ? 1 : (value + 1));
              num = -1;
          }
      }
  }
