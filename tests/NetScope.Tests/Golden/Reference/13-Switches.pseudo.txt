  public static double Area(Shape shape)
  {
      if (shape is Circle circle)
      {
          if (circle.Radius <= 0.0)
          {
              return 0.0;
          }
          return Math.PI * circle.Radius * circle.Radius;
      }
      if (shape is Rect { Width: var width, Height: var height } rect)
      {
          if (width == height)
          {
              return width * width;
          }
          return rect.Width * rect.Height;
      }
      if (shape is Polygon polygon)
      {
          IReadOnlyList<(double, double)> points = polygon.Points;
          if (points != null && points.Count < 3)
          {
              return 0.0;
          }
          return Shoelace(polygon.Points);
      }
      if ((object)shape == null)
      {
          throw new ArgumentNullException("shape");
      }
      throw new NotSupportedException(shape.GetType().Name);
  }

Method   : double Shoelace(IReadOnlyList<ValueTuple<double, double>> points)
Kind     : static, private
RVA      : 0x2B74   Size: 107 bytes   Token: 0x0600003F

[Pseudocode]
  private static double Shoelace(IReadOnlyList<(double X, double Y)> points)
  {
      double num = 0.0;
      for (int i = 0; i < points.Count; i++)
      {
          (double X, double Y) tuple = points[i];
          double item = tuple.X;
          double item2 = tuple.Y;
          (double X, double Y) tuple2 = points[(i + 1) % points.Count];
          double item3 = tuple2.X;
          double item4 = tuple2.Y;
          num += item * item4 - item3 * item2;
      }
      return Math.Abs(num) / 2.0;
  }
